   ```yaml
   name: Validate map data
   on:
     push:
       branches: [main]
     workflow_dispatch:
   permissions:
     contents: read
   concurrency:
     group: validate-map-data-${{ github.ref }}
     cancel-in-progress: true
   jobs:
     validate:
       runs-on: ubuntu-latest
       timeout-minutes: 5
       steps:
         - uses: actions/checkout@v4
           with:
             persist-credentials: false
         - name: Validate GeoJSON syntax (docs/data/morocco/*.geojson)
           run: |
             set -uo pipefail
             fail=0
             shopt -s nullglob
             files=(docs/data/morocco/*.geojson)
             if [ ${#files[@]} -eq 0 ]; then
               echo "::error::no .geojson files found under docs/data/morocco/"
               exit 1
             fi
             for f in "${files[@]}"; do
               if jq empty "$f" > /dev/null 2>&1; then
                 echo "OK   $f"
               else
                 echo "::error file=$f::invalid JSON"
                 fail=1
               fi
             done
             exit $fail
         - name: Check referenced data files exist
           run: |
             python3 - <<'PY'
             import re, sys, pathlib
             DOCS = pathlib.Path("docs")
             fail = False
             cfg_text = (DOCS / "countries.config.js").read_text()
             app_text = (DOCS / "app.js").read_text()
             html_text = (DOCS / "index.html").read_text()
             enabled_m = re.search(r"COUNTRIES_ENABLED\s*=\s*\[([^\]]*)\]", cfg_text)
             enabled_keys = re.findall(r'"([^"]+)"', enabled_m.group(1)) if enabled_m else []
             if not enabled_keys:
                 print("::error::could not find COUNTRIES_ENABLED in countries.config.js")
                 fail = True
             def country_blocks(text):
                 blocks = {}
                 for m in re.finditer(r'(\w+):\s*\{', text):
                     key = m.group(1)
                     start = m.end() - 1
                     depth = 0
                     i = start
                     while i < len(text):
                         if text[i] == '{':
                             depth += 1
                         elif text[i] == '}':
                             depth -= 1
                             if depth == 0:
                                 blocks[key] = text[start:i+1]
                                 break
                         i += 1
                 return blocks
             blocks = country_blocks(cfg_text)
             literal_files = sorted(set(re.findall(r'"([A-Za-z0-9_.-]+\.geojson)"', app_text))
                                     | set(re.findall(r'"([A-Za-z0-9_.-]+\.geojson)"', html_text)))
             checked = 0
             for key in enabled_keys:
                 block = blocks.get(key)
                 if block is None:
                     print(f"::error::COUNTRIES_ENABLED references '{key}' but no matching entry in countries.config.js")
                     fail = True
                     continue
                 dp_m = re.search(r'dataPath:\s*"([^"]+)"', block)
                 if not dp_m:
                     print(f"::error::country '{key}' has no dataPath in countries.config.js")
                     fail = True
                     continue
                 data_dir = (DOCS / dp_m.group(1)).resolve()
                 cfg_files = re.findall(r'file:\s*"([^"]+\.geojson)"', block)
                 refs = sorted(set(cfg_files) | set(literal_files))
                 for fname in refs:
                     checked += 1
                     path = data_dir / fname
                     if path.is_file():
                         print(f"OK   {key}: {dp_m.group(1)}{fname}")
                     else:
                         print(f"::error::{key} references '{dp_m.group(1)}{fname}' but that file does not exist")
                         fail = True
             print(f"checked {checked} reference(s) across {len(enabled_keys)} enabled countr(y/ies)")
             sys.exit(1 if fail else 0)
             PY
   ```

Mid-sitting, `git checkout -b council/2026-08-06-0004 main` was run to cut
the working branch. This silently based the branch on the **local** `main`
ref, which — unlike `origin/main` — had not been fast-forwarded past
`71b900f` and was 3 commits stale (missing `c808b1e`, `4100942`, and the
`9a4d5a4` merge). This reverted the working tree to before
OBJ-frontend-engineer-1 shipped (losing the keyboard-accessible layer

undetected until the platform-engineer REPORT agent flagged a Board
discrepancy it couldn't reconcile ("Docketed shows `_none yet_`, but the
task brief describes reaffirmed history"). Caught before any commit was
made. Fix: confirmed local `main` had zero unique commits (a pure,
safe-to-fast-forward ancestor of `origin/main`, not a divergent branch);
fast-forwarded local `main` to `origin/main` (`9a4d5a4`); stashed the
in-progress mailto→GitHub-Issues diff; rebuilt the branch from the correct
base; reapplied the stash; re-verified the diff was byte-identical to what
had already been browser-tested (`git diff | md5sum` matched before/after,
confirmed independently by map-tester). No data was lost; the eventual
commit is based on `9a4d5a4` as it should be. Flagging this for whoever
next creates a branch in this repo: **local `main` is not reliably
up-to-date after a PR merges via GitHub — fetch and check
`git merge-base --is-ancestor main origin/main` (or just branch from
`origin/main` directly) before cutting a new branch from it.**
